// Lane frame: D along the mid lane, N across it.
const D = [0.85, -0.527], N = [0.527, 0.85];
const off = (dt: number, ds: number) => [dt * N[0] + ds * D[0], dt * N[1] + ds * D[1]];
// Minions are drawn at 1.3× (MS); offsets match ENEMY_OFF in scripts/blueprint.ts.
const MS = 1.3;
const wave = (dir: 1 | -1) => [
  { o: off(-36, 0), m: true }, { o: off(0, 0), m: true }, { o: off(36, 0), m: true },
  { o: off(-23, -44 * dir), m: false }, { o: off(23, -44 * dir), m: false }, { o: off(0, -70 * dir), m: false },
];
const fx = (v: number) => Math.round(v * 10) / 10;

// Minimap (origin 1644,804; 254 px square).
const mm = (x: number, y: number) => [1644 + x, 804 + y];
const blob = (x: number, y: number, r: number, seed: number) => {
  const k = 7, p = [];
  for (let i = 0; i < k; i++) {
    const a = (2 * Math.PI * i) / k + Math.sin(seed + i) * 0.3, rr = r * (0.75 + 0.25 * Math.abs(Math.sin(seed * 3 + i * 1.7)));
    const [px, py] = mm(x + rr * Math.cos(a), y + rr * Math.sin(a) * 0.8);
    p.push(`${fx(px)} ${fx(py)}`);
  }
  return `M${p.join(' L')} Z`;
};
